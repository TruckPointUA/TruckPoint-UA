// ================================ // УСІ КОРИСТУВАЧІ // ================================ let allUsers = []; let friends = {}; let pending = {}; function openUsersModal(){ document.getElementById("usersModal").style.display = "flex"; loadAllUsers(); } function closeUsersModal(){ document.getElementById("usersModal").style.display = "none"; } function loadAllUsers(){ const me = firebase.auth().currentUser; if(!me) return; Promise.all([ db.ref("friends/"+me.uid).once("value"), db.ref("friendRequests/"+me.uid).once("value"), db.ref("users").once("value"), db.ref("usersLocation").once("value") ]).then(([friendsSnap, requestsSnap, usersSnap, locationSnap])=>{ friends = friendsSnap.val() || {}; pending = requestsSnap.val() || {}; const online = {}; locationSnap.forEach(c=>{ online[c.key]=c.val(); }); allUsers=[]; usersSnap.forEach(child=>{ if(child.key===me.uid) return; const d = child.val() || {}; allUsers.push({ uid: child.key, name: ((d.name||"")+" "+(d.surname||"")).trim(), avatar: d.avatar || "avatars/avatar1.png", online: online[child.key]?.online === true }); }); renderUsers(); }); } function renderUsers(){ const list=document.getElementById("usersList"); list.innerHTML=""; allUsers.sort((a,b)=>a.name.localeCompare(b.name)); allUsers.forEach(user=>{ const div=document.createElement("div"); div.className="userItem"; let btn=""; if(friends[user.uid]){ btn=''; }else if(pending[user.uid]){ btn=''; }else{ btn=` `; } div.innerHTML=`
${user.name}
${user.online ? "🟢 Онлайн":"⚪ Офлайн"}
${btn} `; list.appendChild(div); }); } function addFriend(friendUid,friendName){ const me=firebase.auth().currentUser; if(!me) return; const currentName= ( (localStorage.getItem("userName")||"")+ " "+ (localStorage.getItem("userSurname")||"") ).trim(); const currentAvatar= localStorage.getItem("userAvatar") || "avatars/avatar1.png"; db.ref("friendRequests/"+friendUid+"/"+me.uid).set({ uid:me.uid, name:currentName, avatar:currentAvatar, time:Date.now() }).then(()=>{ pending[friendUid]=true; renderUsers(); }); } function filterUsers(){ const text=document.getElementById("userSearch") .value.toLowerCase(); document.querySelectorAll(".userItem").forEach(card=>{ const name=card.querySelector(".userName") .innerText.toLowerCase(); card.style.display= name.includes(text) ?"flex":"none"; }); } document.getElementById("usersModal").onclick=function(e){ if(e.target===this){ closeUsersModal(); } }; function showUserProfile(uid){ db.ref("users/"+uid).once("value").then(s=>{ const d=s.val()||{}; alert( "👤 "+((d.name||"")+" "+(d.surname||"")).trim()+ "\n\nТелефон: "+(d.phone||"не вказано") ); }); } setInterval(()=>{ if(document.getElementById("usersModal").style.display==="flex"){ loadAllUsers(); } },10000);
